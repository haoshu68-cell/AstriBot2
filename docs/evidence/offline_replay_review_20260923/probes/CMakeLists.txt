cmake_minimum_required(VERSION 3.16)
project(replay_review)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp rosbag2_cpp rosbag2_storage rosgraph_msgs nav_msgs geometry_msgs ament_index_cpp)
 find_package(${dep} REQUIRED)
endforeach()
find_package(Qt5 REQUIRED COMPONENTS Widgets)
find_package(nlohmann_json REQUIRED)
find_package(ament_cmake_gtest REQUIRED)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(STATION "${REPO}/ws_robot/src/astribot_operator_station")
add_library(evidence ${STATION}/src/evidence.cpp)
target_include_directories(evidence PUBLIC ${STATION}/include)
target_link_libraries(evidence PUBLIC nlohmann_json::nlohmann_json)
add_executable(replay_stream ${STATION}/src/replay_stream.cpp)
target_include_directories(replay_stream PRIVATE ${STATION}/include)
ament_target_dependencies(replay_stream rclcpp rosbag2_cpp rosbag2_storage rosgraph_msgs)
target_link_libraries(replay_stream nlohmann_json::nlohmann_json)
add_executable(review_probe review_probe.cpp)
ament_target_dependencies(review_probe rclcpp rosbag2_cpp rosbag2_storage nav_msgs geometry_msgs)
target_link_libraries(review_probe nlohmann_json::nlohmann_json)
add_executable(viewer_probe viewer_probe.cpp)
target_include_directories(viewer_probe PRIVATE ${STATION}/src)
ament_target_dependencies(viewer_probe rclcpp rosbag2_cpp rosbag2_storage nav_msgs ament_index_cpp)
target_link_libraries(viewer_probe evidence Qt5::Widgets)
ament_add_gtest(test_evidence ${STATION}/test/test_evidence.cpp)
target_link_libraries(test_evidence evidence)
ament_add_gtest(test_replay_stream ${STATION}/test/test_replay_stream.cpp TIMEOUT 20)
ament_target_dependencies(test_replay_stream rclcpp rosbag2_cpp rosbag2_storage nav_msgs geometry_msgs ament_index_cpp)
