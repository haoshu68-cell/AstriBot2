cmake_minimum_required(VERSION 3.8)
project(astribot_s1_transport_native)
find_package(ament_cmake REQUIRED)
find_package(astribot_navigation_msgs REQUIRED)
find_package(astribot_transport_msgs REQUIRED)
find_package(controller_manager_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(ignition-transport11 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(control_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(astribot_s1_transport_mtc REQUIRED)
find_package(rosidl_default_generators REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "action/HoldResources.action" "action/PlanToHold.action" "srv/RenewHold.srv" DEPENDENCIES geometry_msgs)
add_library(arm_hold_core src/arm_hold.cpp src/resource_authority.cpp src/child_actions.cpp src/resource_journal.cpp src/mtc_plan.cpp src/payload_command.cpp)
target_compile_features(arm_hold_core PUBLIC cxx_std_17)
set_target_properties(arm_hold_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_options(arm_hold_core PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(arm_hold_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(arm_hold_core astribot_navigation_msgs astribot_transport_msgs controller_manager_msgs)
target_link_libraries(arm_hold_core nlohmann_json::nlohmann_json)
add_library(payload_client src/payload_client.cpp)
target_compile_features(payload_client PUBLIC cxx_std_17)
target_compile_options(payload_client PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(payload_client PUBLIC arm_hold_core ignition-transport11::ignition-transport11)
add_executable(hold_executor src/hold_executor.cpp)
target_compile_features(hold_executor PRIVATE cxx_std_17)
target_compile_options(hold_executor PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(hold_executor arm_hold_core)
ament_target_dependencies(hold_executor rclcpp rclcpp_action control_msgs controller_manager_msgs std_msgs)
rosidl_get_typesupport_target(cpp_typesupport_target ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(hold_executor "${cpp_typesupport_target}")
add_library(trajectory_scene_binding STATIC src/scene_binding.cpp)
target_compile_features(trajectory_scene_binding PUBLIC cxx_std_17)
target_include_directories(trajectory_scene_binding PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(trajectory_scene_binding moveit_msgs)
target_link_libraries(trajectory_scene_binding astribot_s1_transport_mtc::transport_scene_signature)
add_executable(trajectory_executor src/hold_executor.cpp)
target_compile_features(trajectory_executor PRIVATE cxx_std_17)
target_compile_definitions(trajectory_executor PRIVATE PLAN_TO_HOLD_EXECUTOR=1)
target_compile_options(trajectory_executor PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(trajectory_executor arm_hold_core trajectory_scene_binding "${cpp_typesupport_target}")
ament_target_dependencies(trajectory_executor rclcpp rclcpp_action control_msgs controller_manager_msgs std_msgs geometry_msgs nav_msgs moveit_msgs astribot_transport_msgs)
install(TARGETS trajectory_executor DESTINATION lib/${PROJECT_NAME})
install(TARGETS hold_executor DESTINATION lib/${PROJECT_NAME})
install(TARGETS arm_hold_core payload_client trajectory_scene_binding EXPORT arm_hold_targets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(arm_hold_targets HAS_LIBRARY_TARGET)
ament_export_dependencies(astribot_navigation_msgs astribot_transport_msgs controller_manager_msgs nlohmann_json ignition-transport11 rosidl_default_runtime astribot_s1_transport_mtc moveit_msgs)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(payload_command_test test/payload_command_test.cpp)
  target_link_libraries(payload_command_test arm_hold_core)
  ament_add_gtest(payload_client_test test/payload_client_test.cpp)
  target_link_libraries(payload_client_test payload_client)
  ament_add_gtest(mtc_plan_test test/mtc_plan_test.cpp)
  target_link_libraries(mtc_plan_test arm_hold_core)
  ament_add_gtest(scene_binding_test test/scene_binding_test.cpp)
  target_link_libraries(scene_binding_test trajectory_scene_binding)
  ament_add_gtest(resource_authority_test test/resource_authority_test.cpp)
  target_link_libraries(resource_authority_test arm_hold_core)
  ament_add_gtest(child_actions_test test/child_actions_test.cpp)
  target_link_libraries(child_actions_test arm_hold_core)
  ament_add_gtest(resource_journal_test test/resource_journal_test.cpp)
  target_link_libraries(resource_journal_test arm_hold_core)
  ament_add_gtest(arm_hold_test test/arm_hold_test.cpp)
  target_link_libraries(arm_hold_test arm_hold_core)
endif()
ament_package()
