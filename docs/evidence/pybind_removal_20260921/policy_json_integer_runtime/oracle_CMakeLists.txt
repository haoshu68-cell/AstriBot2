cmake_minimum_required(VERSION 3.16)
project(recorded_geometry_oracle LANGUAGES CXX)
find_package(Python3 REQUIRED COMPONENTS Interpreter Development)
find_package(pybind11 REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(tinyxml2_vendor REQUIRED)
find_package(TinyXML2 REQUIRED)
find_package(nlohmann_json REQUIRED)
pybind11_add_module(_geometry_native geometry_native.cpp)
target_compile_features(_geometry_native PRIVATE cxx_std_17)
target_compile_options(_geometry_native PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(_geometry_native PRIVATE include)
target_link_libraries(_geometry_native PRIVATE Eigen3::Eigen OpenSSL::Crypto tinyxml2::tinyxml2 nlohmann_json::nlohmann_json)
