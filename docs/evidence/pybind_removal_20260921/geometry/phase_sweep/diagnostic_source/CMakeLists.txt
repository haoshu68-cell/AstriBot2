cmake_minimum_required(VERSION 3.8)
project(astribot_s1_robot_geometry)
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
  # Geometry kernels run inside the sensor freshness budget. Keep explicit
  # Debug/Release selections, but avoid silently shipping unoptimized kernels.
  set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "Build type" FORCE)
endif()
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(tinyxml2_vendor REQUIRED)
find_package(TinyXML2 REQUIRED)
find_package(nlohmann_json REQUIRED)
foreach(dependency rclcpp ament_index_cpp sensor_msgs std_msgs rcl_interfaces moveit_msgs astribot_navigation_msgs)
  find_package(${dependency} REQUIRED)
endforeach()
add_library(geometry_core INTERFACE)
target_include_directories(geometry_core INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_features(geometry_core INTERFACE cxx_std_17)
target_link_libraries(geometry_core INTERFACE Eigen3::Eigen OpenSSL::Crypto tinyxml2::tinyxml2 nlohmann_json::nlohmann_json)
install(TARGETS geometry_core EXPORT geometry_core_targets)
ament_export_targets(geometry_core_targets)
ament_export_dependencies(Eigen3 OpenSSL tinyxml2_vendor TinyXML2 nlohmann_json)
option(ASTRIBOT_GEOMETRY_BUILD_PYTHON_COMPAT "Build temporary compatibility bindings for unmigrated consumers" ON)
if(ASTRIBOT_GEOMETRY_BUILD_PYTHON_COMPAT)
  find_package(ament_cmake_python REQUIRED)
  find_package(Python3 REQUIRED COMPONENTS Interpreter Development)
  find_package(pybind11 REQUIRED)
  ament_python_install_package(${PROJECT_NAME})
  ament_get_python_install_dir(PYTHON_INSTALL_DIR)
  pybind11_add_module(_geometry_native src/geometry_native.cpp)
  target_include_directories(_geometry_native PRIVATE include)
  target_compile_features(_geometry_native PRIVATE cxx_std_17)
  target_compile_options(_geometry_native PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(_geometry_native PRIVATE geometry_core)
  install(TARGETS _geometry_native LIBRARY DESTINATION "${PYTHON_INSTALL_DIR}/${PROJECT_NAME}")
endif()
install(DIRECTORY include/ DESTINATION include OPTIONAL)
add_executable(geometry_state_timing src/geometry_state_node.cpp)
target_link_libraries(geometry_state_timing geometry_core)
target_compile_options(geometry_state_timing PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(geometry_state_timing rclcpp ament_index_cpp sensor_msgs std_msgs rcl_interfaces moveit_msgs astribot_navigation_msgs)
install(TARGETS geometry_state_timing DESTINATION lib/${PROJECT_NAME})
ament_export_include_directories(include)
find_package(geometry_msgs REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
ament_export_dependencies(geometry_msgs nav2_costmap_2d)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  enable_testing()
  add_executable(filled_collision_test test/filled_collision_test.cpp)
  target_include_directories(filled_collision_test PRIVATE include)
  ament_target_dependencies(filled_collision_test geometry_msgs nav2_costmap_2d)
  add_test(NAME filled_collision COMMAND filled_collision_test)
  add_executable(fusion_snapshot_test test/fusion_snapshot_test.cpp)
  target_include_directories(fusion_snapshot_test PRIVATE include)
  add_test(NAME fusion_snapshot COMMAND fusion_snapshot_test)
  add_executable(geometry_state_core_test test/geometry_state_core_test.cpp)
  target_link_libraries(geometry_state_core_test geometry_core)
  add_test(NAME geometry_state_core COMMAND geometry_state_core_test)
  foreach(test_target filled_collision_test fusion_snapshot_test geometry_state_core_test)
    target_compile_features(${test_target} PRIVATE cxx_std_17)
    target_compile_options(${test_target} PRIVATE -UNDEBUG)
  endforeach()
  add_test(NAME cpp_geometry_differential COMMAND ${CMAKE_COMMAND} -E env
    "PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR}:$ENV{PYTHONPATH}"
    ${Python3_EXECUTABLE} -m pytest -q
    ${CMAKE_CURRENT_SOURCE_DIR}/test/test_cpp_geometry_kernels.py
    ${CMAKE_CURRENT_SOURCE_DIR}/test/test_cpp_robot_model.py)
  add_test(NAME geometry_state_cpp_protocol COMMAND ${CMAKE_COMMAND} -E env
    "PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR}:$ENV{PYTHONPATH}"
    "GEOMETRY_STATE_CPP=$<TARGET_FILE:geometry_state_timing>" "ROS_DOMAIN_ID=115" "ROS_LOCALHOST_ONLY=1"
    ${Python3_EXECUTABLE} -m pytest -q ${CMAKE_CURRENT_SOURCE_DIR}/test/test_geometry_state_cpp_protocol.py)
endif()
ament_package()
