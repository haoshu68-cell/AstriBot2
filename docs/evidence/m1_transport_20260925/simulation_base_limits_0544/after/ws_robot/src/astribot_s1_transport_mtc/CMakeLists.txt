cmake_minimum_required(VERSION 3.16)
project(astribot_s1_transport_mtc)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_get_python_install_dir(PYTHON_INSTALL_DIR)
_ament_cmake_python_register_environment_hook()
add_subdirectory(native_scene)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(ompl REQUIRED)
find_package(moveit_planners_ompl REQUIRED)
find_package(control_msgs REQUIRED)
find_package(moveit_task_constructor_core REQUIRED)
find_package(astribot_s1_manipulation REQUIRED)
find_package(astribot_transport_msgs REQUIRED)
find_package(astribot_eigen_vendor REQUIRED)
find_package(moveit_ros_planning_interface REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(octomap REQUIRED)
find_package(moveit_msgs REQUIRED)
add_library(transport_scene_signature SHARED src/canonical_octomap.cpp src/canonical_scene.cpp)
target_include_directories(transport_scene_signature PUBLIC
 $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
 $<INSTALL_INTERFACE:include>)
target_include_directories(transport_scene_signature PRIVATE ${OCTOMAP_INCLUDE_DIRS})
target_link_libraries(transport_scene_signature PRIVATE ${OCTOMAP_LIBRARIES})
ament_target_dependencies(transport_scene_signature PUBLIC moveit_msgs)
add_executable(execution_guard src/execution_guard.cpp)
target_include_directories(execution_guard PRIVATE include)
ament_target_dependencies(execution_guard rclcpp control_msgs astribot_transport_msgs tf2_ros tf2_geometry_msgs)
add_executable(mtc_planner src/mtc_planner.cpp src/payload_transition.cpp)
target_include_directories(mtc_planner PRIVATE include)
target_link_libraries(mtc_planner transport_scene_signature)
ament_target_dependencies(mtc_planner rclcpp rclcpp_action moveit_task_constructor_core
 astribot_s1_manipulation astribot_transport_msgs moveit_ros_planning_interface)
astribot_target_eigen(mtc_planner)
install(TARGETS mtc_planner execution_guard DESTINATION lib/${PROJECT_NAME})
install(TARGETS transport_scene_signature EXPORT transport_scene_signature_targets
 ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(FILES include/astribot_s1_transport_mtc/canonical_octomap.hpp
 include/astribot_s1_transport_mtc/canonical_scene.hpp
 include/astribot_s1_transport_mtc/base_motion_limits.hpp
 DESTINATION include/${PROJECT_NAME})
ament_export_targets(transport_scene_signature_targets HAS_LIBRARY_TARGET)
ament_export_dependencies(moveit_msgs)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(joint_trajectory_controller REQUIRED)
  ament_add_gtest(trajectory_time_scaling_test test/trajectory_time_scaling_test.cpp)
  ament_target_dependencies(trajectory_time_scaling_test astribot_s1_manipulation joint_trajectory_controller)
  astribot_target_eigen(trajectory_time_scaling_test)
  ament_add_gtest(canonical_scene_test test/canonical_scene_test.cpp)
  target_link_libraries(canonical_scene_test transport_scene_signature)
  ament_add_gtest(payload_transition_test test/payload_transition_test.cpp src/payload_transition.cpp)
  target_include_directories(payload_transition_test PRIVATE include ${OCTOMAP_INCLUDE_DIRS})
  target_link_libraries(payload_transition_test transport_scene_signature ${OCTOMAP_LIBRARIES})
  ament_target_dependencies(payload_transition_test astribot_s1_manipulation astribot_transport_msgs moveit_ros_planning_interface)
  astribot_target_eigen(payload_transition_test)
  enable_testing()
  add_executable(execution_guard_test test/execution_guard_test.cpp)
  target_include_directories(execution_guard_test PRIVATE include)
  add_test(NAME execution_guard_boundaries COMMAND execution_guard_test)
  add_executable(joint_margin_test test/joint_margin_test.cpp)
  target_include_directories(joint_margin_test PRIVATE include)
  add_test(NAME joint_planning_margin COMMAND joint_margin_test)
endif()
ament_package()
