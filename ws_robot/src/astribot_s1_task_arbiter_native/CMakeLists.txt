cmake_minimum_required(VERSION 3.16)
project(astribot_s1_task_arbiter_native)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(ament_cmake REQUIRED)
find_package(astribot_navigation_zones REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(action_msgs REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(astribot_navigation_msgs REQUIRED)
find_package(astribot_s1_path_tracking REQUIRED)
add_executable(task_arbiter_cpp src/task_arbiter_node.cpp)
target_compile_features(task_arbiter_cpp PRIVATE cxx_std_17)
target_compile_options(task_arbiter_cpp PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(task_arbiter_cpp PRIVATE include ${astribot_s1_path_tracking_INCLUDE_DIRS})
ament_target_dependencies(task_arbiter_cpp rclcpp rclcpp_action action_msgs nav2_msgs astribot_navigation_msgs)
target_link_libraries(task_arbiter_cpp astribot_navigation_zones::zone_core)
ament_target_dependencies(task_arbiter_cpp std_msgs)
install(TARGETS task_arbiter_cpp DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(ownership_test test/ownership_test.cpp)
  target_compile_features(ownership_test PRIVATE cxx_std_17)
  target_compile_options(ownership_test PRIVATE -UNDEBUG)
  target_include_directories(ownership_test PRIVATE include)
  add_test(NAME ownership_test COMMAND ownership_test)
endif()
ament_package()
