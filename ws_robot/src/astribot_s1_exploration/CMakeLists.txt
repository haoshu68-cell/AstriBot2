cmake_minimum_required(VERSION 3.16)
project(astribot_s1_exploration)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(astribot_navigation_zones REQUIRED)
find_package(astribot_slam_msgs REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(astribot_navigation_msgs REQUIRED)
find_package(astribot_operator_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_components REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(astribot_autonomy_core REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(std_srvs REQUIRED)
add_library(${PROJECT_NAME}_components SHARED src/frontier_explorer_node.cpp src/exploration_coordinator_node.cpp)
target_include_directories(${PROJECT_NAME}_components PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME}_components astribot_operator_msgs astribot_navigation_msgs rclcpp rclcpp_components sensor_msgs nav_msgs geometry_msgs std_msgs visualization_msgs rcl_interfaces tf2 tf2_ros tf2_geometry_msgs nav2_msgs rclcpp_action std_srvs)
target_link_libraries(${PROJECT_NAME}_components astribot_navigation_zones::zone_core astribot_autonomy_core::exploration_domain nlohmann_json::nlohmann_json)
rclcpp_components_register_nodes(${PROJECT_NAME}_components "astribot_s1_autonomy::FrontierExplorerNode" "astribot_s1_autonomy::ExplorationCoordinatorNode")
add_executable(frontier_explorer_node src/frontier_explorer_main.cpp)
target_link_libraries(frontier_explorer_node ${PROJECT_NAME}_components)
ament_target_dependencies(frontier_explorer_node rclcpp rclcpp_components sensor_msgs nav_msgs geometry_msgs std_msgs visualization_msgs rcl_interfaces tf2 tf2_ros tf2_geometry_msgs nav2_msgs rclcpp_action std_srvs)
install(TARGETS frontier_explorer_node DESTINATION lib/${PROJECT_NAME})
add_executable(exploration_coordinator_node src/exploration_coordinator_main.cpp)
target_link_libraries(exploration_coordinator_node ${PROJECT_NAME}_components)
ament_target_dependencies(exploration_coordinator_node rclcpp rclcpp_components sensor_msgs nav_msgs geometry_msgs std_msgs visualization_msgs rcl_interfaces tf2 tf2_ros tf2_geometry_msgs nav2_msgs rclcpp_action std_srvs)
install(TARGETS exploration_coordinator_node DESTINATION lib/${PROJECT_NAME})
install(TARGETS ${PROJECT_NAME}_components EXPORT export_${PROJECT_NAME} ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch config rviz DESTINATION share/${PROJECT_NAME})
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_export_include_directories(include)
ament_export_dependencies(astribot_navigation_zones astribot_operator_msgs astribot_navigation_msgs rclcpp rclcpp_components sensor_msgs nav_msgs geometry_msgs std_msgs visualization_msgs rcl_interfaces tf2 tf2_ros tf2_geometry_msgs astribot_autonomy_core nav2_msgs rclcpp_action std_srvs)
add_executable(mapping_session_node src/mapping_session_node.cpp)
target_include_directories(mapping_session_node PRIVATE include)
ament_target_dependencies(mapping_session_node rclcpp rcl_interfaces nav_msgs std_msgs std_srvs astribot_slam_msgs)
target_link_libraries(mapping_session_node astribot_navigation_zones::zone_core yaml-cpp nlohmann_json::nlohmann_json OpenSSL::Crypto)
install(TARGETS mapping_session_node DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_exploration_end test/test_exploration_end.cpp TIMEOUT 20
    ENV ROS_DOMAIN_ID=229 ROS_LOCALHOST_ONLY=1)
  target_link_libraries(test_exploration_end ${PROJECT_NAME}_components)
  ament_target_dependencies(test_exploration_end rclcpp std_srvs)
  ament_add_gtest(test_mapping_session test/test_mapping_session.cpp TIMEOUT 45
    ENV ROS_DOMAIN_ID=230 ROS_LOCALHOST_ONLY=1)
  target_include_directories(test_mapping_session PRIVATE include)
  ament_target_dependencies(test_mapping_session rclcpp rcl_interfaces nav_msgs std_msgs std_srvs astribot_slam_msgs)
  target_link_libraries(test_mapping_session astribot_navigation_zones::zone_core yaml-cpp nlohmann_json::nlohmann_json OpenSSL::Crypto)
endif()
ament_package()
