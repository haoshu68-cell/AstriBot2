cmake_minimum_required(VERSION 3.16)
project(astribot_s1_exploration)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(astribot_navigation_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_components REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(astribot_autonomy_core REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(std_srvs REQUIRED)
add_library(${PROJECT_NAME}_components SHARED src/frontier_explorer_node.cpp src/exploration_coordinator_node.cpp)
target_include_directories(${PROJECT_NAME}_components PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME}_components astribot_navigation_msgs rclcpp rclcpp_components sensor_msgs nav_msgs geometry_msgs std_msgs visualization_msgs rcl_interfaces tf2 tf2_ros tf2_geometry_msgs nav2_msgs rclcpp_action std_srvs)
target_link_libraries(${PROJECT_NAME}_components astribot_autonomy_core::exploration_domain)
rclcpp_components_register_nodes(${PROJECT_NAME}_components "astribot_s1_autonomy::FrontierExplorerNode" "astribot_s1_autonomy::ExplorationCoordinatorNode")
add_executable(frontier_explorer_node src/frontier_explorer_main.cpp)
target_link_libraries(frontier_explorer_node ${PROJECT_NAME}_components)
ament_target_dependencies(frontier_explorer_node rclcpp rclcpp_components sensor_msgs nav_msgs geometry_msgs std_msgs visualization_msgs rcl_interfaces tf2 tf2_ros tf2_geometry_msgs nav2_msgs rclcpp_action std_srvs)
install(TARGETS frontier_explorer_node DESTINATION lib/${PROJECT_NAME})
add_executable(exploration_coordinator_node src/exploration_coordinator_main.cpp)
target_link_libraries(exploration_coordinator_node ${PROJECT_NAME}_components)
ament_target_dependencies(exploration_coordinator_node rclcpp rclcpp_components sensor_msgs nav_msgs geometry_msgs std_msgs visualization_msgs rcl_interfaces tf2 tf2_ros tf2_geometry_msgs nav2_msgs rclcpp_action std_srvs)
install(TARGETS exploration_coordinator_node DESTINATION lib/${PROJECT_NAME})
install(TARGETS ${PROJECT_NAME}_components EXPORT export_${PROJECT_NAME} ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_export_include_directories(include)
ament_export_dependencies(astribot_navigation_msgs rclcpp rclcpp_components sensor_msgs nav_msgs geometry_msgs std_msgs visualization_msgs rcl_interfaces tf2 tf2_ros tf2_geometry_msgs astribot_autonomy_core nav2_msgs rclcpp_action std_srvs)
ament_package()
