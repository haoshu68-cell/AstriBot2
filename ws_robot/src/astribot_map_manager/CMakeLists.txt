cmake_minimum_required(VERSION 3.16)
project(astribot_map_manager)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp std_msgs nav_msgs nav2_msgs tf2_ros astribot_operator_msgs)
 find_package(${dep} REQUIRED)
endforeach()
find_package(astribot_s1_exploration REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(spdlog REQUIRED)
add_library(map_catalog src/catalog.cpp src/voxel_activation.cpp)
target_include_directories(map_catalog PUBLIC include ${astribot_s1_exploration_INCLUDE_DIRS})
target_link_libraries(map_catalog PUBLIC nlohmann_json::nlohmann_json OpenSSL::Crypto yaml-cpp)
add_executable(voxel_session_adapter src/voxel_session_adapter.cpp)
ament_target_dependencies(voxel_session_adapter rclcpp std_msgs nav_msgs nav2_msgs tf2_ros astribot_operator_msgs)
target_link_libraries(voxel_session_adapter map_catalog spdlog::spdlog)
add_executable(map_manager src/map_manager.cpp)
ament_target_dependencies(map_manager rclcpp std_msgs nav_msgs astribot_operator_msgs)
target_link_libraries(map_manager map_catalog spdlog::spdlog)
add_executable(fake_map_adapter src/fake_map_adapter.cpp)
ament_target_dependencies(fake_map_adapter rclcpp std_msgs astribot_operator_msgs)
target_link_libraries(fake_map_adapter nlohmann_json::nlohmann_json)
install(TARGETS voxel_session_adapter map_manager fake_map_adapter DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
 find_package(ament_cmake_gtest REQUIRED)
 ament_add_gtest(test_voxel_activation test/test_voxel_activation.cpp)
 target_link_libraries(test_voxel_activation map_catalog)
 ament_add_gtest(test_voxel_adapter test/test_voxel_adapter.cpp TIMEOUT 20 ENV ROS_DOMAIN_ID=215 ROS_LOCALHOST_ONLY=1)
 ament_target_dependencies(test_voxel_adapter rclcpp std_msgs nav_msgs nav2_msgs tf2_ros astribot_operator_msgs)
 target_link_libraries(test_voxel_adapter map_catalog spdlog::spdlog)
 ament_add_gtest(test_catalog test/test_catalog.cpp)
 target_link_libraries(test_catalog map_catalog)
 ament_add_gtest(test_map_manager test/test_map_manager.cpp TIMEOUT 35 ENV ROS_DOMAIN_ID=216 ROS_LOCALHOST_ONLY=1)
 ament_target_dependencies(test_map_manager rclcpp std_msgs nav_msgs astribot_operator_msgs)
 target_link_libraries(test_map_manager map_catalog spdlog::spdlog)
endif()
ament_package()
