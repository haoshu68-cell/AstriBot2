cmake_minimum_required(VERSION 3.16)
project(astribot_s1_navigation_policy_native LANGUAGES CXX)

find_package(ament_cmake REQUIRED)
find_package(astribot_navigation_zones REQUIRED)
option(ASTRIBOT_BUILD_PYBIND "Build the migration-only Python bridge" ON)
if(ASTRIBOT_BUILD_PYBIND)
  find_package(ament_cmake_python REQUIRED)
  find_package(Python3 REQUIRED COMPONENTS Interpreter Development)
  find_package(pybind11 REQUIRED)
endif()
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(astribot_navigation_msgs REQUIRED)
find_package(astribot_perception_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(astribot_s1_robot_geometry REQUIRED)
find_package(OpenSSL REQUIRED)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "Build type" FORCE)
endif()

add_library(navigation_math STATIC
  src/navigation_math.cpp
  src/cmd_vel_body_to_world_core.cpp)
target_include_directories(navigation_math PUBLIC include)
target_compile_features(navigation_math PUBLIC cxx_std_17)
target_compile_options(navigation_math PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
set_target_properties(navigation_math PROPERTIES POSITION_INDEPENDENT_CODE ON)

if(ASTRIBOT_BUILD_PYBIND)
  pybind11_add_module(_navigation_math_native src/navigation_math_pybind.cpp)
  target_link_libraries(_navigation_math_native PRIVATE navigation_math)
  target_include_directories(_navigation_math_native PRIVATE include)
  target_compile_features(_navigation_math_native PRIVATE cxx_std_17)
  target_compile_options(_navigation_math_native PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(cmd_vel_body_to_world_cpp src/cmd_vel_body_to_world_node.cpp)
target_link_libraries(cmd_vel_body_to_world_cpp navigation_math)
target_include_directories(cmd_vel_body_to_world_cpp PRIVATE include)
target_compile_features(cmd_vel_body_to_world_cpp PRIVATE cxx_std_17)
target_compile_options(cmd_vel_body_to_world_cpp PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(cmd_vel_body_to_world_cpp
  rclcpp rcl_interfaces geometry_msgs nav_msgs)

add_executable(costmap_scan_cpp src/costmap_scan_node.cpp)
target_link_libraries(costmap_scan_cpp navigation_math)
target_include_directories(costmap_scan_cpp PRIVATE include)
target_compile_features(costmap_scan_cpp PRIVATE cxx_std_17)
target_compile_options(costmap_scan_cpp PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(costmap_scan_cpp rclcpp sensor_msgs)

add_library(policy_profile STATIC src/policy_profile.cpp)
target_include_directories(policy_profile PUBLIC include)
target_link_libraries(policy_profile PUBLIC nlohmann_json::nlohmann_json)
target_compile_options(policy_profile PRIVATE -Wall -Wextra -Wpedantic)
add_executable(envelope_coordinator_cpp src/envelope_coordinator_node.cpp)
target_compile_features(envelope_coordinator_cpp PRIVATE cxx_std_17)
target_compile_options(envelope_coordinator_cpp PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(envelope_coordinator_cpp policy_profile)
ament_target_dependencies(envelope_coordinator_cpp
  rclcpp geometry_msgs nav_msgs tf2 tf2_ros tf2_geometry_msgs astribot_navigation_msgs)

add_library(final_protection_core STATIC src/final_protection_core.cpp)
target_include_directories(final_protection_core PUBLIC include)
target_link_libraries(final_protection_core PUBLIC navigation_math nlohmann_json::nlohmann_json OpenSSL::Crypto)
ament_target_dependencies(final_protection_core PUBLIC astribot_navigation_msgs astribot_s1_robot_geometry)
target_compile_options(final_protection_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(navigation_constraint_cpp src/navigation_constraint_node.cpp)
target_link_libraries(navigation_constraint_cpp astribot_navigation_zones::zone_core)
target_link_libraries(navigation_constraint_cpp final_protection_core policy_profile)
ament_target_dependencies(navigation_constraint_cpp
  rclcpp ament_index_cpp geometry_msgs nav_msgs nav2_msgs sensor_msgs std_msgs tf2_ros astribot_s1_robot_geometry astribot_navigation_msgs astribot_perception_msgs)
target_compile_options(navigation_constraint_cpp PRIVATE -Wall -Wextra -Wpedantic)

if(ASTRIBOT_BUILD_PYBIND)
  ament_python_install_package(${PROJECT_NAME})
  ament_get_python_install_dir(PYTHON_INSTALL_DIR)
  install(TARGETS _navigation_math_native
    LIBRARY DESTINATION "${PYTHON_INSTALL_DIR}/${PROJECT_NAME}")
endif()
install(TARGETS cmd_vel_body_to_world_cpp
  RUNTIME DESTINATION lib/${PROJECT_NAME})
install(TARGETS costmap_scan_cpp
  RUNTIME DESTINATION lib/${PROJECT_NAME})
install(TARGETS envelope_coordinator_cpp
  RUNTIME DESTINATION lib/${PROJECT_NAME})
install(TARGETS navigation_constraint_cpp
  RUNTIME DESTINATION lib/${PROJECT_NAME} COMPONENT navigation_constraint)

if(BUILD_TESTING)
  enable_testing()
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(scan_timing_test test/scan_timing_test.cpp)

  target_include_directories(publication_lease_test PRIVATE include)
  ament_add_gtest(projection_capability_test test/projection_capability_test.cpp)
  target_include_directories(projection_capability_test PRIVATE include)
  ament_add_gtest(navigation_constraint_ros_test test/navigation_constraint_ros_test.cpp)

  ament_target_dependencies(navigation_constraint_ros_test rclcpp ament_index_cpp geometry_msgs nav_msgs nav2_msgs sensor_msgs std_msgs tf2_ros astribot_s1_robot_geometry astribot_navigation_msgs astribot_perception_msgs)
  add_executable(navigation_math_test test/navigation_math_test.cpp)

  add_executable(policy_profile_probe test/policy_profile_probe.cpp)
