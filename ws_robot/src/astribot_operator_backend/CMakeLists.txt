cmake_minimum_required(VERSION 3.16)
project(astribot_operator_backend)
set(CMAKE_CXX_STANDARD 17)
include_directories(include)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp rclcpp_action astribot_operator_msgs astribot_navigation_msgs astribot_transport_msgs moveit_msgs nav2_msgs std_msgs std_srvs sensor_msgs nav_msgs)
 find_package(${dep} REQUIRED)
endforeach()
find_package(nlohmann_json REQUIRED)
add_executable(fake_navigation src/fake_navigation.cpp)
ament_target_dependencies(fake_navigation rclcpp rclcpp_action nav2_msgs)
add_executable(mapping_runtime src/mapping_runtime.cpp)
ament_target_dependencies(mapping_runtime rclcpp std_msgs std_srvs nav_msgs)
target_link_libraries(mapping_runtime nlohmann_json::nlohmann_json)
add_executable(operator_backend src/operator_backend.cpp)
ament_target_dependencies(operator_backend rclcpp rclcpp_action astribot_operator_msgs astribot_navigation_msgs astribot_transport_msgs moveit_msgs nav2_msgs std_msgs std_srvs sensor_msgs nav_msgs)
target_link_libraries(operator_backend nlohmann_json::nlohmann_json)
install(TARGETS operator_backend mapping_runtime fake_navigation DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
 find_package(ament_cmake_gtest REQUIRED)
 ament_add_gtest(test_arm_execution_transaction test/test_arm_execution_transaction.cpp)
 ament_add_gtest(test_arm_preview test/test_arm_preview.cpp TIMEOUT 15 ENV ROS_DOMAIN_ID=214 ROS_LOCALHOST_ONLY=1)
 ament_target_dependencies(test_arm_preview rclcpp astribot_transport_msgs moveit_msgs sensor_msgs)
 target_link_libraries(test_arm_preview nlohmann_json::nlohmann_json)
 ament_add_gtest(test_mapping_runtime test/test_mapping_runtime.cpp TIMEOUT 15 ENV ROS_DOMAIN_ID=220 ROS_LOCALHOST_ONLY=1)
 ament_target_dependencies(test_mapping_runtime rclcpp std_msgs std_srvs nav_msgs)
 target_link_libraries(test_mapping_runtime nlohmann_json::nlohmann_json)
 ament_add_gtest(test_backend test/test_backend.cpp TIMEOUT 40 ENV ROS_DOMAIN_ID=221 ROS_LOCALHOST_ONLY=1)
 ament_target_dependencies(test_backend rclcpp rclcpp_action astribot_operator_msgs astribot_navigation_msgs astribot_transport_msgs moveit_msgs nav2_msgs std_msgs std_srvs sensor_msgs nav_msgs)
 target_link_libraries(test_backend nlohmann_json::nlohmann_json)
endif()
ament_package()
