cmake_minimum_required(VERSION 3.16)
project(astribot_s1_perception_native)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
if(BUILD_TESTING)
  enable_testing()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
endif()
include(cmake/map_odom.cmake)
add_executable(map_domain_relay src/map_domain_relay.cpp)
target_compile_features(map_domain_relay PRIVATE cxx_std_17)
target_compile_options(map_domain_relay PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(map_domain_relay rclcpp nav_msgs)
install(TARGETS map_domain_relay DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  add_test(NAME map_relay_protocol COMMAND ${Python3_EXECUTABLE} -m pytest -q
    ${CMAKE_CURRENT_SOURCE_DIR}/test/test_map_relay.py)
  set_tests_properties(map_relay_protocol PROPERTIES TIMEOUT 180 RUN_SERIAL TRUE
    ENVIRONMENT "ROS_LOCALHOST_ONLY=1;RMW_IMPLEMENTATION=rmw_fastrtps_cpp;MAP_RELAY_CPP=$<TARGET_FILE:map_domain_relay>;MAP_RELAY_EVIDENCE=${CMAKE_CURRENT_BINARY_DIR}/map_relay_ros")
endif()
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_package()
