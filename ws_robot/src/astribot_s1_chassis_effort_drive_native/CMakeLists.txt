cmake_minimum_required(VERSION 3.8)
project(astribot_s1_chassis_effort_drive_native)

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)

if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "Build type" FORCE)
endif()

add_library(wheel_math STATIC src/wheel_math.cpp)
target_include_directories(wheel_math PUBLIC include)
target_compile_features(wheel_math PUBLIC cxx_std_17)
target_compile_options(wheel_math PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(wheel_math PROPERTIES POSITION_INDEPENDENT_CODE ON)

install(DIRECTORY include/ DESTINATION include)
ament_export_include_directories(include)

add_executable(omni_effort_drive_cpp src/omni_effort_drive_node.cpp)
target_link_libraries(omni_effort_drive_cpp wheel_math)
target_include_directories(omni_effort_drive_cpp PRIVATE include)
target_compile_features(omni_effort_drive_cpp PRIVATE cxx_std_17)
target_compile_options(omni_effort_drive_cpp PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(omni_effort_drive_cpp rclcpp geometry_msgs sensor_msgs std_msgs)
install(TARGETS omni_effort_drive_cpp
  RUNTIME DESTINATION lib/${PROJECT_NAME})

if(BUILD_TESTING)
  enable_testing()
  add_executable(wheel_math_test test/wheel_math_test.cpp)
  target_link_libraries(wheel_math_test PRIVATE wheel_math)
  target_include_directories(wheel_math_test PRIVATE include)
  target_compile_features(wheel_math_test PRIVATE cxx_std_17)
  target_compile_options(wheel_math_test PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
  add_test(NAME wheel_math COMMAND wheel_math_test)
  add_executable(wheel_reversal_test test/wheel_reversal_test.cpp)
  target_link_libraries(wheel_reversal_test PRIVATE wheel_math)
  target_compile_features(wheel_reversal_test PRIVATE cxx_std_17)
  add_test(NAME wheel_reversal COMMAND wheel_reversal_test)
  add_executable(wheel_math_benchmark test/wheel_math_benchmark.cpp)
  target_link_libraries(wheel_math_benchmark wheel_math)
  target_include_directories(wheel_math_benchmark PRIVATE include)
  target_compile_features(wheel_math_benchmark PRIVATE cxx_std_17)
  target_compile_options(wheel_math_benchmark PRIVATE -Wall -Wextra -Wpedantic)
endif()

ament_package()
