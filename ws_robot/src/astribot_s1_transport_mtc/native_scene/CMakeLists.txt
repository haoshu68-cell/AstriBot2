cmake_minimum_required(VERSION 3.16)
project(astribot_transport_scene_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(octomap REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter Development)
find_package(pybind11 REQUIRED)
add_library(transport_scene_core STATIC ../src/canonical_octomap.cpp)
set_target_properties(transport_scene_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(transport_scene_core PUBLIC ../include ${OCTOMAP_INCLUDE_DIRS})
target_link_libraries(transport_scene_core PUBLIC ${OCTOMAP_LIBRARIES})
target_compile_options(transport_scene_core PRIVATE -Wall -Wextra -Wpedantic)
pybind11_add_module(_transport_scene_native ../src/transport_scene_pybind.cpp)
target_link_libraries(_transport_scene_native PRIVATE transport_scene_core)
if(NOT DEFINED PYTHON_INSTALL_DIR)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  set(PYTHON_INSTALL_DIR "local/lib/python${Python3_VERSION_MAJOR}.${Python3_VERSION_MINOR}/dist-packages")
endif()
install(TARGETS _transport_scene_native LIBRARY DESTINATION ${PYTHON_INSTALL_DIR})
if(BUILD_TESTING OR CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
enable_testing()
add_executable(canonical_octomap_test ../test/canonical_octomap_test.cpp)
target_link_libraries(canonical_octomap_test PRIVATE transport_scene_core)
add_test(NAME canonical_octomap_boundaries COMMAND canonical_octomap_test)
endif()
