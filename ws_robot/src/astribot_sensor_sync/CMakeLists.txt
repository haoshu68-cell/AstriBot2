cmake_minimum_required(VERSION 3.8)
project(astribot_sensor_sync)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rosidl_default_generators REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "msg/TriggerEdge.msg" "msg/SensorTiming.msg" "msg/SyncStatus.msg")
rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} "rosidl_typesupport_cpp")
add_executable(sync_monitor src/sync_monitor.cpp)
target_include_directories(sync_monitor PRIVATE include)
ament_target_dependencies(sync_monitor rclcpp)
target_link_libraries(sync_monitor "${typesupport}")
install(TARGETS sync_monitor DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(sync_core_test test/sync_core_test.cpp)
  target_include_directories(sync_core_test PRIVATE include)
endif()
ament_export_include_directories(include)
ament_export_dependencies(rosidl_default_runtime)
ament_package()
