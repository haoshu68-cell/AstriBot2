cmake_minimum_required(VERSION 3.16)
project(astribot_operator_station)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp rclcpp_action rcl_interfaces std_msgs std_srvs nav2_msgs rosbag2_cpp rosbag2_storage rviz_common rviz_rendering rviz_default_plugins geometry_msgs visualization_msgs astribot_operator_msgs pluginlib)
  find_package(${dep} REQUIRED)
endforeach()
find_package(rosgraph_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets Concurrent)
find_package(nlohmann_json REQUIRED)
find_package(spdlog REQUIRED)
add_library(evidence src/evidence.cpp)
set_target_properties(evidence PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(evidence PUBLIC include)
target_link_libraries(evidence PUBLIC nlohmann_json::nlohmann_json)
add_executable(diagnostics_recorder src/recorder.cpp)
target_link_libraries(diagnostics_recorder evidence spdlog::spdlog)
ament_target_dependencies(diagnostics_recorder rclcpp rcl_interfaces std_msgs std_srvs rosbag2_cpp rosbag2_storage)
add_executable(fake_exploration_backend src/fake_exploration_backend.cpp)
ament_target_dependencies(fake_exploration_backend rclcpp std_msgs std_srvs astribot_operator_msgs)
target_link_libraries(fake_exploration_backend nlohmann_json::nlohmann_json)
add_executable(replay_stream src/replay_stream.cpp)
target_include_directories(replay_stream PRIVATE include)
ament_target_dependencies(replay_stream rclcpp rosbag2_cpp rosbag2_storage rosgraph_msgs)
target_link_libraries(replay_stream nlohmann_json::nlohmann_json)
add_executable(replay_viewer src/replay_viewer.cpp)
target_link_libraries(replay_viewer evidence Qt5::Widgets)
ament_target_dependencies(replay_viewer ament_index_cpp rosbag2_cpp rosbag2_storage rclcpp nav_msgs)
add_executable(inspect_incident src/inspect.cpp)
target_link_libraries(inspect_incident evidence)
add_library(operator_panel SHARED src/map_page.cpp include/astribot_operator_station/map_page.hpp src/workstation_panel.cpp include/astribot_operator_station/workstation_panel.hpp src/exploration_controls.cpp include/astribot_operator_station/exploration_controls.hpp src/panel.cpp src/loop_route_panel.cpp src/route_point_tool.cpp
 include/astribot_operator_station/panel.hpp include/astribot_operator_station/loop_route_panel.hpp)
set_target_properties(operator_panel PROPERTIES AUTOMOC ON)
target_include_directories(operator_panel PRIVATE include)
target_link_libraries(operator_panel evidence Qt5::Widgets Qt5::Concurrent)
ament_target_dependencies(operator_panel ament_index_cpp rclcpp rclcpp_action std_msgs std_srvs nav2_msgs rviz_common rviz_default_plugins geometry_msgs visualization_msgs astribot_operator_msgs pluginlib)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
install(TARGETS diagnostics_recorder inspect_incident fake_exploration_backend replay_stream replay_viewer DESTINATION lib/${PROJECT_NAME})
install(TARGETS operator_panel LIBRARY DESTINATION lib)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
