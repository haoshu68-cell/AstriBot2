cmake_minimum_required(VERSION 3.16)
project(astribot_s1_manipulation_perception)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(astribot_perception_msgs REQUIRED)
find_package(astribot_transport_msgs REQUIRED)
find_package(astribot_s1_manipulation REQUIRED)
find_package(control_msgs REQUIRED)
find_package(ompl REQUIRED)
find_package(moveit_planners_ompl REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
find_package(nlohmann_json REQUIRED)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
add_library(inference_contract src/inference_contract.cpp src/worker_process.cpp)
target_include_directories(inference_contract PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(inference_contract sensor_msgs)
target_link_libraries(inference_contract Threads::Threads)
add_executable(manipulation_perception_server src/manipulation_perception_server.cpp)
target_link_libraries(manipulation_perception_server inference_contract nlohmann_json::nlohmann_json OpenSSL::Crypto)
ament_target_dependencies(manipulation_perception_server rclcpp rclcpp_action astribot_perception_msgs tf2)
add_library(pick_planning_client src/pick_planning_contract.cpp src/pick_planning_client.cpp)
target_include_directories(pick_planning_client PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(pick_planning_client rclcpp rclcpp_action astribot_perception_msgs astribot_transport_msgs tf2 tf2_geometry_msgs)
install(TARGETS pick_planning_client EXPORT pick_planning_targets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
ament_export_targets(pick_planning_targets HAS_LIBRARY_TARGET)
ament_export_dependencies(rclcpp rclcpp_action astribot_perception_msgs astribot_transport_msgs tf2 tf2_geometry_msgs)
add_library(known_box_grasp_mapping src/known_box_grasp_mapping.cpp)
target_include_directories(known_box_grasp_mapping PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(known_box_grasp_mapping pick_planning_client astribot_s1_manipulation::astribot_s1_manipulation_gripper)
ament_target_dependencies(known_box_grasp_mapping astribot_s1_manipulation)
install(TARGETS known_box_grasp_mapping EXPORT box_mapping_targets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
ament_export_targets(box_mapping_targets HAS_LIBRARY_TARGET)
ament_export_dependencies(astribot_s1_manipulation control_msgs ompl moveit_planners_ompl)
add_library(single_box_request_source src/single_box_request.cpp src/single_box_request_source.cpp)
target_include_directories(single_box_request_source PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(single_box_request_source pick_planning_client OpenSSL::Crypto nlohmann_json::nlohmann_json ${OpenCV_LIBS})
ament_target_dependencies(single_box_request_source rclcpp tf2_ros astribot_perception_msgs sensor_msgs)
install(TARGETS single_box_request_source EXPORT rgbd_source_targets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
ament_export_targets(rgbd_source_targets HAS_LIBRARY_TARGET)
ament_export_dependencies(tf2_ros OpenCV OpenSSL nlohmann_json)
install(TARGETS inference_contract manipulation_perception_server ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
