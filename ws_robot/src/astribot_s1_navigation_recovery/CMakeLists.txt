cmake_minimum_required(VERSION 3.8)
project(astribot_s1_navigation_recovery)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(astribot_eigen_vendor REQUIRED)
foreach(dep astribot_navigation_msgs astribot_slam_msgs astribot_s1_robot_geometry astribot_s1_path_tracking
    astribot_bridge_msgs nav2_core nav2_costmap_2d nav2_util nav2_behavior_tree behaviortree_cpp_v3
    diagnostic_msgs geometry_msgs nav_msgs std_msgs pluginlib rclcpp rclcpp_lifecycle tf2 tf2_ros tf2_geometry_msgs)
  find_package(${dep} REQUIRED)
endforeach()
add_library(${PROJECT_NAME} SHARED src/navigation_start_assessment.cpp src/departure_planner.cpp src/departure_controller.cpp)
target_include_directories(${PROJECT_NAME} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME} astribot_navigation_msgs astribot_slam_msgs astribot_s1_robot_geometry
  astribot_s1_path_tracking astribot_bridge_msgs nav2_core nav2_costmap_2d nav2_util geometry_msgs nav_msgs std_msgs
  pluginlib rclcpp rclcpp_lifecycle tf2 tf2_ros tf2_geometry_msgs)
target_link_libraries(${PROJECT_NAME} astribot_s1_robot_geometry::geometry_core)
astribot_target_eigen(${PROJECT_NAME})
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
add_library(astribot_navigation_recovery_bt_node SHARED src/navigation_start_bt.cpp)
target_compile_definitions(astribot_navigation_recovery_bt_node PRIVATE BT_PLUGIN_EXPORT)
ament_target_dependencies(astribot_navigation_recovery_bt_node astribot_s1_path_tracking astribot_navigation_msgs diagnostic_msgs geometry_msgs
  nav2_behavior_tree behaviortree_cpp_v3 rclcpp)
install(TARGETS ${PROJECT_NAME} astribot_navigation_recovery_bt_node ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
install(FILES plugins.xml README.md DESTINATION share/${PROJECT_NAME})
ament_export_include_directories(include)
ament_export_libraries(${PROJECT_NAME})
ament_export_dependencies(astribot_s1_path_tracking astribot_navigation_msgs astribot_slam_msgs astribot_s1_robot_geometry nav2_core nav2_costmap_2d)
if(BUILD_TESTING)
  enable_testing()
  add_executable(departure_path_test test/departure_path_test.cpp)
  target_include_directories(departure_path_test PRIVATE include)
  target_link_libraries(departure_path_test astribot_s1_robot_geometry::geometry_core)
  target_compile_options(departure_path_test PRIVATE -UNDEBUG)
  ament_target_dependencies(departure_path_test astribot_navigation_msgs astribot_slam_msgs geometry_msgs nav_msgs nav2_costmap_2d)
  add_test(NAME departure_path COMMAND departure_path_test)
  add_executable(departure_controller_test test/departure_controller_test.cpp)
  target_include_directories(departure_controller_test PRIVATE include)
  target_link_libraries(departure_controller_test ${PROJECT_NAME})
  ament_target_dependencies(departure_controller_test astribot_s1_path_tracking astribot_navigation_msgs astribot_slam_msgs
    astribot_bridge_msgs nav2_core nav2_costmap_2d nav_msgs pluginlib rclcpp rclcpp_lifecycle tf2 tf2_ros)
  add_test(NAME departure_controller COMMAND departure_controller_test)
  add_executable(navigation_start_assessment_test test/navigation_start_assessment_test.cpp)
  target_include_directories(navigation_start_assessment_test PRIVATE include)
  target_link_libraries(navigation_start_assessment_test ${PROJECT_NAME})
  target_compile_options(navigation_start_assessment_test PRIVATE -UNDEBUG)
  ament_target_dependencies(navigation_start_assessment_test astribot_s1_path_tracking astribot_navigation_msgs astribot_slam_msgs
    astribot_bridge_msgs nav2_core nav2_costmap_2d nav_msgs pluginlib rclcpp rclcpp_lifecycle tf2 tf2_ros)
  add_test(NAME navigation_start_assessment COMMAND navigation_start_assessment_test)
  add_executable(recovery_plugin_load_test test/recovery_plugin_load_test.cpp)
  ament_target_dependencies(recovery_plugin_load_test pluginlib nav2_core)
  add_test(NAME recovery_plugin_load COMMAND recovery_plugin_load_test)
  add_executable(navigation_start_bt_test test/navigation_start_bt_test.cpp)
  target_compile_options(navigation_start_bt_test PRIVATE -UNDEBUG)

    astribot_navigation_msgs nav2_behavior_tree behaviortree_cpp_v3)
  add_test(NAME navigation_start_bt COMMAND navigation_start_bt_test
    $<TARGET_FILE:astribot_navigation_recovery_bt_node>
    ${CMAKE_CURRENT_SOURCE_DIR}/../astribot_s1_navigation/behavior_trees/navigate_to_pose_precise_goal.xml)
endif()
