cmake_minimum_required(VERSION 3.16)
project(astribot_s1_perception_components)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_components REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(astribot_autonomy_core REQUIRED)
find_package(pcl_conversions REQUIRED)
add_library(${PROJECT_NAME}_components SHARED src/pointcloud_slice_scan_node.cpp)
target_include_directories(${PROJECT_NAME}_components PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME}_components rclcpp rclcpp_components sensor_msgs nav_msgs geometry_msgs std_msgs visualization_msgs rcl_interfaces tf2 tf2_ros tf2_geometry_msgs pcl_conversions)
target_link_libraries(${PROJECT_NAME}_components astribot_autonomy_core::perception_geometry)
rclcpp_components_register_nodes(${PROJECT_NAME}_components "astribot_s1_autonomy::PointcloudSliceScanNode")
add_executable(pointcloud_slice_scan_node src/pointcloud_slice_scan_main.cpp)
target_link_libraries(pointcloud_slice_scan_node ${PROJECT_NAME}_components)
ament_target_dependencies(pointcloud_slice_scan_node rclcpp rclcpp_components sensor_msgs nav_msgs geometry_msgs std_msgs visualization_msgs rcl_interfaces tf2 tf2_ros tf2_geometry_msgs pcl_conversions)
install(TARGETS pointcloud_slice_scan_node DESTINATION lib/${PROJECT_NAME})
find_package(PCL REQUIRED COMPONENTS common filters)
target_include_directories(${PROJECT_NAME}_components SYSTEM PUBLIC ${PCL_INCLUDE_DIRS})
target_link_libraries(${PROJECT_NAME}_components ${PCL_LIBRARIES})
find_path(LIVOX_MSG_INCLUDE_DIR NAMES livox_ros_driver2/msg/custom_msg.hpp PATH_SUFFIXES include/livox_ros_driver2 include)
find_library(LIVOX_MSG_TYPESUPPORT_CPP NAMES livox_ros_driver2__rosidl_typesupport_cpp)
find_library(LIVOX_MSG_GENERATOR_C NAMES livox_ros_driver2__rosidl_generator_c)
if(LIVOX_MSG_INCLUDE_DIR AND LIVOX_MSG_TYPESUPPORT_CPP AND LIVOX_MSG_GENERATOR_C)
  add_executable(livox_custom_to_pc2_node src/livox_custom_to_pc2_main.cpp src/livox_custom_to_pc2_node.cpp)
  target_include_directories(livox_custom_to_pc2_node PRIVATE include ${LIVOX_MSG_INCLUDE_DIR})
  ament_target_dependencies(livox_custom_to_pc2_node rclcpp sensor_msgs rcl_interfaces)
  target_link_libraries(livox_custom_to_pc2_node astribot_autonomy_core::perception_geometry ${LIVOX_MSG_TYPESUPPORT_CPP} ${LIVOX_MSG_GENERATOR_C})
  install(TARGETS livox_custom_to_pc2_node DESTINATION lib/${PROJECT_NAME})
else()
  message(STATUS "Livox CustomMsg unavailable: converter executable not built")
endif()
install(TARGETS ${PROJECT_NAME}_components EXPORT export_${PROJECT_NAME} ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_export_include_directories(include)
ament_export_dependencies(rclcpp rclcpp_components sensor_msgs nav_msgs geometry_msgs std_msgs visualization_msgs rcl_interfaces tf2 tf2_ros tf2_geometry_msgs astribot_autonomy_core pcl_conversions)
ament_package()
