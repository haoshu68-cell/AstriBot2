cmake_minimum_required(VERSION 3.16)
project(astribot_s1_perception_components)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(astribot_eigen_vendor REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_components REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(shape_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(astribot_autonomy_core REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(pluginlib REQUIRED)
add_library(astribot_observed_ray_layer SHARED src/observed_ray_obstacle_layer.cpp)
target_include_directories(astribot_observed_ray_layer PRIVATE include)
ament_target_dependencies(astribot_observed_ray_layer nav2_costmap_2d pluginlib sensor_msgs)
pluginlib_export_plugin_description_file(nav2_costmap_2d costmap_plugins.xml)
install(TARGETS astribot_observed_ray_layer LIBRARY DESTINATION lib)
add_library(${PROJECT_NAME}_components SHARED src/pointcloud_slice_scan_node.cpp)
target_include_directories(${PROJECT_NAME}_components PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME}_components rclcpp rclcpp_components sensor_msgs nav_msgs geometry_msgs std_msgs moveit_msgs visualization_msgs rcl_interfaces tf2 tf2_ros tf2_geometry_msgs pcl_conversions)
target_link_libraries(${PROJECT_NAME}_components astribot_autonomy_core::perception_geometry)
rclcpp_components_register_nodes(${PROJECT_NAME}_components "astribot_s1_autonomy::PointcloudSliceScanNode")
add_executable(pointcloud_slice_scan_node src/pointcloud_slice_scan_main.cpp)
target_link_libraries(pointcloud_slice_scan_node ${PROJECT_NAME}_components)
ament_target_dependencies(pointcloud_slice_scan_node rclcpp rclcpp_components sensor_msgs nav_msgs geometry_msgs std_msgs moveit_msgs visualization_msgs rcl_interfaces tf2 tf2_ros tf2_geometry_msgs pcl_conversions)
install(TARGETS pointcloud_slice_scan_node DESTINATION lib/${PROJECT_NAME})
find_package(PCL REQUIRED COMPONENTS common filters)
target_include_directories(${PROJECT_NAME}_components SYSTEM PUBLIC ${PCL_INCLUDE_DIRS})
target_link_libraries(${PROJECT_NAME}_components ${PCL_LIBRARIES})
install(TARGETS ${PROJECT_NAME}_components EXPORT export_${PROJECT_NAME} ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_export_include_directories(include)
ament_export_dependencies(astribot_eigen_vendor rclcpp rclcpp_components sensor_msgs nav_msgs geometry_msgs std_msgs moveit_msgs visualization_msgs rcl_interfaces tf2 tf2_ros tf2_geometry_msgs astribot_autonomy_core pcl_conversions)
astribot_target_eigen(${PROJECT_NAME}_components)
astribot_target_eigen(pointcloud_slice_scan_node)
if(BUILD_TESTING)
  enable_testing()
  add_executable(attached_body_filter_test test/attached_body_filter_test.cpp)
  target_include_directories(attached_body_filter_test PRIVATE include)
  ament_target_dependencies(attached_body_filter_test geometry_msgs shape_msgs)
  astribot_target_eigen(attached_body_filter_test)
  add_test(NAME attached_body_filter COMMAND attached_body_filter_test)
  add_executable(attachment_filter_confirmation_test test/attachment_filter_confirmation_test.cpp)
  target_include_directories(attachment_filter_confirmation_test PRIVATE include)
  add_test(NAME attachment_filter_confirmation COMMAND attachment_filter_confirmation_test)
endif()
ament_package()
