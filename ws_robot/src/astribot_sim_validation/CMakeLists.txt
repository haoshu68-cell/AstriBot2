cmake_minimum_required(VERSION 3.16)
project(astribot_sim_validation)
set(CMAKE_CXX_STANDARD 17)
include_directories(include)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp rosgraph_msgs controller_manager_msgs sensor_msgs geometry_msgs nav_msgs std_msgs std_srvs astribot_operator_msgs astribot_navigation_msgs)
 find_package(${dep} REQUIRED)
endforeach()
find_package(nlohmann_json REQUIRED)
add_executable(simulation_probe src/simulation_probe.cpp)
ament_target_dependencies(simulation_probe rclcpp rosgraph_msgs controller_manager_msgs sensor_msgs nav_msgs)
target_link_libraries(simulation_probe nlohmann_json::nlohmann_json)
add_executable(transport_acceptance src/transport_acceptance.cpp)
ament_target_dependencies(transport_acceptance rclcpp std_msgs astribot_operator_msgs)
target_link_libraries(transport_acceptance nlohmann_json::nlohmann_json)
add_executable(transport_session src/transport_session.cpp)
ament_target_dependencies(transport_session rclcpp std_msgs std_srvs astribot_navigation_msgs)
target_link_libraries(transport_session nlohmann_json::nlohmann_json)
install(TARGETS transport_acceptance transport_session simulation_probe DESTINATION lib/${PROJECT_NAME})
add_executable(transport_fault_acceptance src/transport_fault_acceptance.cpp)
ament_target_dependencies(transport_fault_acceptance rclcpp std_msgs nav_msgs sensor_msgs geometry_msgs astribot_operator_msgs astribot_navigation_msgs)
target_link_libraries(transport_fault_acceptance nlohmann_json::nlohmann_json)
install(TARGETS transport_fault_acceptance DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
 find_package(ament_cmake_gtest REQUIRED)
 ament_add_gtest(test_session_evidence test/test_session_evidence.cpp)
 target_link_libraries(test_session_evidence nlohmann_json::nlohmann_json)
 ament_add_gtest(test_fault_evidence test/test_fault_evidence.cpp)
 target_link_libraries(test_fault_evidence nlohmann_json::nlohmann_json)
endif()
ament_package()
