cmake_minimum_required(VERSION 3.16)
project(astribot_navigation_zones)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(nlohmann_json REQUIRED)
foreach(dep rclcpp nav2_costmap_2d nav2_util pluginlib std_msgs nav_msgs geometry_msgs tf2_ros astribot_operator_msgs)
 find_package(${dep} REQUIRED)
endforeach()
add_library(zone_core SHARED src/geometry.cpp src/snapshot.cpp src/store.cpp)
target_include_directories(zone_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(zone_core PUBLIC nlohmann_json::nlohmann_json)
add_executable(zone_server src/zone_server.cpp)
target_link_libraries(zone_server zone_core)
ament_target_dependencies(zone_server rclcpp std_msgs nav_msgs astribot_operator_msgs)
install(TARGETS zone_server DESTINATION lib/${PROJECT_NAME})
add_library(zone_layer SHARED src/zone_layer.cpp)
target_link_libraries(zone_layer zone_core)
ament_target_dependencies(zone_layer rclcpp nav2_costmap_2d nav2_util pluginlib std_msgs tf2_ros)
pluginlib_export_plugin_description_file(nav2_costmap_2d plugins.xml)
install(TARGETS zone_layer LIBRARY DESTINATION lib)
install(TARGETS zone_core EXPORT zone_targets LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(zone_targets HAS_LIBRARY_TARGET)
ament_export_dependencies(nlohmann_json rclcpp std_msgs)
if(BUILD_TESTING)
 find_package(ament_cmake_gtest REQUIRED)
 ament_add_gtest(test_zone_ros test/test_ros.cpp TIMEOUT 30 ENV ROS_DOMAIN_ID=218 ROS_LOCALHOST_ONLY=1)
 target_link_libraries(test_zone_ros zone_core zone_layer)
 ament_target_dependencies(test_zone_ros rclcpp std_msgs nav_msgs astribot_operator_msgs nav2_costmap_2d nav2_util tf2_ros)
 ament_add_gtest(test_zone_geometry test/test_geometry.cpp)
 target_link_libraries(test_zone_geometry zone_core)
 ament_add_gtest(test_zone_state test/test_state.cpp)
 target_link_libraries(test_zone_state zone_core)
endif()
ament_package()
