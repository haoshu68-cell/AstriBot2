cmake_minimum_required(VERSION 3.16)
project(astribot_route_executor)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp rclcpp_action nav2_msgs std_msgs astribot_operator_msgs)
 find_package(${dep} REQUIRED)
endforeach()
find_package(nlohmann_json REQUIRED)
add_executable(loop_route_executor src/loop_route_executor.cpp)
ament_target_dependencies(loop_route_executor rclcpp rclcpp_action nav2_msgs std_msgs astribot_operator_msgs)
target_link_libraries(loop_route_executor nlohmann_json::nlohmann_json)
install(TARGETS loop_route_executor DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
 find_package(ament_cmake_gtest REQUIRED)
 ament_add_gtest(test_loop_route test/test_loop_route.cpp TIMEOUT 45 ENV ROS_DOMAIN_ID=228 ROS_LOCALHOST_ONLY=1)
 ament_target_dependencies(test_loop_route rclcpp rclcpp_action nav2_msgs std_msgs astribot_operator_msgs)
 target_link_libraries(test_loop_route nlohmann_json::nlohmann_json)
endif()
ament_package()
