cmake_minimum_required(VERSION 3.8)
project(astribot_s1_payload_state)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(astribot_payload_msgs REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(OpenSSL REQUIRED)
add_library(payload_state_core src/ledger.cpp src/contract.cpp src/journal.cpp src/consumer.cpp)
set_target_properties(payload_state_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(payload_state_core PUBLIC cxx_std_17)
target_compile_options(payload_state_core PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(payload_state_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(payload_state_core astribot_payload_msgs moveit_msgs)
target_link_libraries(payload_state_core nlohmann_json::nlohmann_json OpenSSL::Crypto)
add_executable(payload_state src/payload_state_node.cpp)
target_link_libraries(payload_state payload_state_core)
ament_target_dependencies(payload_state rclcpp moveit_msgs)
install(TARGETS payload_state DESTINATION lib/${PROJECT_NAME})
install(TARGETS payload_state_core EXPORT payload_state_targets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(payload_state_targets HAS_LIBRARY_TARGET)
ament_export_dependencies(astribot_payload_msgs moveit_msgs nlohmann_json OpenSSL)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(payload_state_test test/ledger_test.cpp)
  target_link_libraries(payload_state_test payload_state_core)
  ament_add_gtest(payload_consumer_test test/consumer_test.cpp)
  target_link_libraries(payload_consumer_test payload_state_core)
endif()
ament_package()
